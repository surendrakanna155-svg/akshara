DETAILED DATABASE SCHEMA - CORE SYSTEM

====================================================
1. TABLE: schools
====================================================

Purpose:
Master school information.

Columns:

id UUID PRIMARY KEY
school_name VARCHAR(255)
school_code VARCHAR(50)
email VARCHAR(255)
phone VARCHAR(20)
address TEXT
city VARCHAR(100)
state VARCHAR(100)
country VARCHAR(100)
status VARCHAR(50)

created_at TIMESTAMP
updated_at TIMESTAMP

Indexes:
school_code

====================================================
2. TABLE: school_branches
====================================================

id UUID PRIMARY KEY
school_id UUID

branch_name VARCHAR(255)
branch_code VARCHAR(50)

address TEXT

created_at TIMESTAMP

Relationships:
school_id -> schools.id

====================================================
3. TABLE: academic_years
====================================================

id UUID PRIMARY KEY
school_id UUID

year_name VARCHAR(50)

start_date DATE
end_date DATE

status VARCHAR(50)

====================================================
4. TABLE: users
====================================================

id UUID PRIMARY KEY
school_id UUID

email VARCHAR(255)
mobile VARCHAR(20)

password_hash TEXT

first_name VARCHAR(100)
last_name VARCHAR(100)

profile_photo_url TEXT

language_preference VARCHAR(50)

status VARCHAR(50)

created_at TIMESTAMP
updated_at TIMESTAMP

Indexes:

email
mobile
school_id

====================================================
5. TABLE: roles
====================================================

6. TABLE: permissions
====================================================

7. TABLE: role_permissions

8. TABLE: user_roles

9. TABLE: students
====================================================

Indexes:

school_id
admission_number
