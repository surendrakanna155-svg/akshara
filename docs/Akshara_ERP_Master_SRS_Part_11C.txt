ACADEMIC OPERATIONS DATABASE SCHEMA

====================================================
1. TABLE: student_attendance
====================================================

id UUID PRIMARY KEY
school_id UUID
academic_year_id UUID

student_id UUID
class_id UUID
section_id UUID

attendance_date DATE

status VARCHAR(20)

remarks TEXT

created_at TIMESTAMP

Indexes:
student_id
attendance_date

====================================================
2. TABLE: staff_attendance
====================================================

id UUID PRIMARY KEY

school_id UUID
staff_id UUID

attendance_date DATE

check_in_time TIMESTAMP
check_out_time TIMESTAMP

status VARCHAR(20)

attendance_method VARCHAR(50)

====================================================
3. TABLE: face_recognition_logs
====================================================

id UUID PRIMARY KEY

captured_at TIMESTAMP

====================================================

====================================================
